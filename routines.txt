package routines;

import java.util.*;
import java.text.SimpleDateFormat;

public class MigUtils {

    /** Clé cible (UUID). */
    public static String uuid() { return java.util.UUID.randomUUID().toString(); }

    /** Horodatage courant. */
    public static java.util.Date now() { return new java.util.Date(); }

    /** Dates Amplitude (jj/MM/aa, jj/MM/aaaa, aaaa-MM-jj). */
    public static java.util.Date parseAmpDate(String s){
        if(s==null||s.trim().isEmpty()) return null;
        String[] fmts={"dd/MM/yy","dd/MM/yyyy","yyyy-MM-dd"};
        for(String f:fmts){ try{ return new SimpleDateFormat(f).parse(s.trim()); }catch(Exception e){} }
        return null;
    }

    /** Surcharge : si Oracle renvoie deja une Date (colonne typee Date), on la renvoie telle quelle.
     *  Evite l'erreur "parseAmpDate(String) is not applicable for the arguments (Date)". */
    public static java.util.Date parseAmpDate(java.util.Date d){ return d; }

    /** Année d'une date (pour graduation_year). */
    public static Integer yearOf(java.util.Date d){
        if(d==null) return null;
        Calendar c=Calendar.getInstance(); c.setTime(d);
        return c.get(Calendar.YEAR);
    }

    /** Statut employé dérivé de MRRX (RA = radié -> TERMINATED). */
    public static String statusFromMrrx(String mrrx){
        return (mrrx!=null && mrrx.trim().equalsIgnoreCase("RA")) ? "TERMINATED" : "ACTIVE";
    }

    /** 1er e-mail trouvé dans des champs libres (ADR1..ADR4/BPOS). */
    public static String extractEmail(String... fields){
        if(fields==null) return null;
        java.util.regex.Pattern p=java.util.regex.Pattern.compile("[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\\.[A-Za-z]{2,}");
        for(String f:fields){ if(f!=null && f.indexOf('@')>0){ java.util.regex.Matcher m=p.matcher(f); if(m.find()) return m.group(); } }
        return null;
    }

    /** Ne garde que les chiffres d'une chaine (supprime espaces, . / ? ! - , etc.). */
    public static String digitsOnly(String s){
        return s==null? "" : s.replaceAll("[^0-9]","");
    }

    /** Normalise UN numéro camerounais à partir d'une séquence de chiffres deja extraite.
     *  - si la sequence commence par 6 : on prend 9 chiffres, prefixe +237
     *  - sinon : on prend 8 chiffres, on ajoute un 6 devant, prefixe +237
     *  Renvoie null si pas assez de chiffres. */
    public static String normalizeCmPhone(String digits){
        if(digits==null || digits.length()==0) return null;
        if(digits.charAt(0)=='6'){
            if(digits.length()<9) return null;
            return "+237"+digits.substring(0,9);
        } else {
            if(digits.length()<8) return null;
            return "+237"+"6"+digits.substring(0,8);
        }
    }

    /** Longueur d'un 1er numero (9 si commence par 6, sinon 8) — sert a savoir ou commence le 2e. */
    private static int firstLen(String digits){
        return (digits!=null && digits.length()>0 && digits.charAt(0)=='6') ? 9 : 8;
    }

    /** Numero de telephone principal (1er numero normalise, format +237...). */
    public static String phonePrincipal(String ntel){
        String d=digitsOnly(ntel);
        return normalizeCmPhone(d);
    }

    /** Numero du contact d'urgence :
     *  - s'il y a un 2e numero (chaine de chiffres > longueur du 1er), on le normalise ;
     *  - sinon on reconduit le 1er numero. Toujours au format +237... */
    public static String phoneUrgence(String ntel){
        String d=digitsOnly(ntel);
        if(d==null || d.length()==0) return null;
        int fl=firstLen(d);
        if(d.length()>fl){
            String rest=d.substring(fl);
            String second=normalizeCmPhone(rest);
            if(second!=null) return second;
        }
        return normalizeCmPhone(d); // un seul numero -> on le reconduit
    }

    /** Echappe une valeur pour l'inclure dans une chaine JSON. */
    public static String jsonEsc(String v){
        if(v==null) return "";
        StringBuilder b=new StringBuilder();
        for(int i=0;i<v.length();i++){
            char c=v.charAt(i);
            switch(c){
                case '"':  b.append("\\\""); break;
                case '\\': b.append("\\\\"); break;
                case '\n': b.append("\\n"); break;
                case '\r': b.append("\\r"); break;
                case '\t': b.append("\\t"); break;
                default:   b.append(c);
            }
        }
        return b.toString();
    }

    /** Adresse au format Fluxoon en CHAINE JSON : {city,state,street,country,countryCode}.
     *  Renvoie null si vide (colonne jsonb NULL). La colonne doit etre castee ::jsonb (voir guide). */
    public static String jsonAddress(String adr1,String adr2,String adr3,String adr4,String ville,String bpos){
        StringBuilder street=new StringBuilder();
        for(String sline:new String[]{adr1,adr2,adr3,adr4}){
            if(sline!=null && sline.trim().length()>0 && sline.indexOf('@')<0){
                if(street.length()>0) street.append(", ");
                street.append(sline.trim());
            }
        }
        String city = (ville!=null && ville.trim().length()>0)? ville.trim() : "";
        String streetStr = street.toString();
        if(bpos!=null && bpos.trim().length()>0 && bpos.indexOf('@')<0){
            streetStr = streetStr.isEmpty()? ("BP "+bpos.trim()) : (streetStr+" - BP "+bpos.trim());
        }
        if(city.isEmpty() && streetStr.isEmpty()) return null;
        return "{\"city\":\""+jsonEsc(city)+"\",\"state\":\"\",\"street\":\""+jsonEsc(streetStr)
              +"\",\"country\":\"Cameroon\",\"countryCode\":\"CM\"}";
    }

    /** Contact d'urgence en CHAINE JSON (array) : [{"phone","name","relationship"}]. null si vide. */
    public static String jsonEmergency(String phoneUrg){
        if(phoneUrg==null || phoneUrg.trim().isEmpty()) return null;
        return "[{\"phone\":\""+jsonEsc(phoneUrg)+"\",\"name\":\"\",\"relationship\":\"\"}]";
    }
}
